{{/* Relative href to another version's copy of the page being rendered. Takes a dict: target (an entry from data/versions.json), cur (this build's version string), versions (the whole list). The home layout renders with the overview page's context, and only `page` reflects the output file the link resolves against. - RelPermalink is relative to baseURL, so it leaves out the version's path prefix. Every version but the latest has one, and it is a level to climb. */}} {{ $depth := 0 }} {{ range split (strings.Trim page.RelPermalink "/") "/" }} {{ if ne . "" }}{{ $depth = add $depth 1 }}{{ end }} {{ end }} {{ $curPrefix := "" }} {{ range .versions }} {{ if eq .version $.cur }}{{ $curPrefix = .path }}{{ end }} {{ end }} {{ if ne $curPrefix "" }}{{ $depth = add $depth 1 }}{{ end }} {{ $prefix := cond (eq .target.path "") "" (printf "%s/" .target.path) }} {{ $href := printf "%s%s%s" (strings.Repeat $depth "../") $prefix (strings.TrimPrefix "/" page.RelPermalink) }} {{ return $href }}
